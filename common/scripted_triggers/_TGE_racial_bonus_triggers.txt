# Racial building bonuses - triggers.
# Per bonus: a non-oppressed majority (coexisting, integrated or primary Anbennar pop modifier) gives the full bonus,
# a non-oppressed minority (coexisting or integrated, small or large) gives the half one, if the province has one of the race's buildings.
# Applied by common/scripted_effects/TGE_racial_bonus_effects.txt. The majority/minority triggers also drive the mine prospecting bonus (mine_marcin.1).

tge_racial_dwarven_forgecraft_majority_trigger = {
	OR = {
		has_province_modifier = dwarven_majority_coexisting
		has_province_modifier = dwarven_majority_integrated
		has_province_modifier = dwarven_majority_primary
	}
}

tge_racial_dwarven_forgecraft_minority_trigger = {
	OR = {
		has_province_modifier = dwarven_minority_coexisting_small
		has_province_modifier = dwarven_minority_coexisting_large
		has_province_modifier = dwarven_minority_integrated_small
		has_province_modifier = dwarven_minority_integrated_large
	}
}

tge_racial_dwarven_forgecraft_buildings_trigger = {
	OR = {
		has_building = metalworking_workshop
		has_building = cannon_foundry
		has_building = furnace
		has_building = weapons
	}
}
