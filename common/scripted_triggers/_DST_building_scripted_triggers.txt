no_ongoing_construction_triggers = {
    NOT = { has_construction = core }
    NOT = { has_construction = culture }
    NOT = { has_construction = building }
    NOT = { has_construction = missionary }
    NOT = { has_construction = great_project }
    NOT = { has_construction = canal }
}

meet_basic_province_requirements_trigger = {
    is_state_core = ROOT
    controlled_by = ROOT
    has_siege = no
}

manufactory_production_limiter_trigger = {
	if = {
		limit = {
			owner = { has_country_flag = building_selection_15_prod_flag }
		}
		base_production = 15
	}
	else_if = {
		limit = {
			owner = { has_country_flag = building_selection_10_prod_flag }
		}
		base_production = 10
	}
	else_if = {
		limit = {
			owner = { has_country_flag = building_selection_5_prod_flag }
		}
		base_production = 5
	}
	else = {
		
	}
}

forcelimit_development_limiter_trigger = {
	if = {
		limit = {
			owner = { has_country_flag = building_selection_30_development_flag }
		}
		development = 30
	}
	else_if = {
		limit = {
			owner = { has_country_flag = building_selection_20_development_flag }
		}
		development = 20
	}
	else_if = {
		limit = {
			owner = { has_country_flag = building_selection_10_development_flag }
		}
		development = 10
	}
	else = {
		
	}
}

# AI road builder (ai_building_events.10), province scope, ROOT = the building country.
# One trigger per road: the province is big enough and has nothing that good yet.
ai_road_site_trigger = {
    NOT = { local_autonomy = 50 }
    NOT = { has_province_flag = ai_road_queued }
    meet_basic_province_requirements_trigger = yes
    no_ongoing_construction_triggers = yes
}
ai_road_site_railroad_trigger = {
    development = 15
    NOT = { has_building = railroad }
    ROOT = { dip_tech = 27 }
    ai_road_site_trigger = yes
}
ai_road_site_macadam_trigger = {
    development = 15
    NOT = { has_building = macadam_road }
    NOT = { has_building = railroad }
    ROOT = { dip_tech = 23 }
    ai_road_site_trigger = yes
}
ai_road_site_royal_trigger = {
    development = 15
    NOT = { has_building = royal_road }
    NOT = { has_building = macadam_road }
    NOT = { has_building = railroad }
    ROOT = { dip_tech = 17 }
    ai_road_site_trigger = yes
}
ai_road_site_gravel_trigger = {
    development = 10
    NOT = { has_building = gravel_road }
    NOT = { has_building = royal_road }
    NOT = { has_building = macadam_road }
    NOT = { has_building = railroad }
    ROOT = { dip_tech = 10 }
    ai_road_site_trigger = yes
}
ai_road_site_early_trigger = {
    development = 5
    NOT = { has_building = early_road }
    NOT = { has_building = gravel_road }
    NOT = { has_building = royal_road }
    NOT = { has_building = macadam_road }
    NOT = { has_building = railroad }
    ROOT = { dip_tech = 3 }
    ai_road_site_trigger = yes
}
ai_road_site_any_trigger = {
    OR = {
        ai_road_site_railroad_trigger = yes
        ai_road_site_macadam_trigger = yes
        ai_road_site_royal_trigger = yes
        ai_road_site_gravel_trigger = yes
        ai_road_site_early_trigger = yes
    }
}
