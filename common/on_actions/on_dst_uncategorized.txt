on_startup = {
	events = {
		tge_activate_goods.1
	}
	# Strategic resources: first run, or first load of a save from before the current system (on_startup also fires on load)
	tge_resource_system_init = yes
}

# Strategic resources: humans every month, AI every year. tge_recalc_country also covers subjects.
on_monthly_pulse = {
	if = {
		limit = { ai = no }
		tge_recalc_and_notify = yes
	}
	# Racial building bonuses: humans every month
	if = {
		limit = { ai = no }
		tge_racial_bonus_refresh = yes
	}
}

on_yearly_pulse = {
	if = {
		limit = {
			ai = yes
			is_subject = no
		}
		tge_recalc_country = yes
	}
	# Racial building bonuses: AI every year
	if = {
		limit = { ai = yes }
		tge_racial_bonus_refresh = yes
	}
}

#When adding effects directly in on_actions always consider if it would be best to do so through a scripted effect. on_actions with a lot of script in them are harder to get an overview of.

# ROOT = province, FROM = country
# Used by Trade Goods Expanded
on_trade_good_changed = {
	# Strategic resources
	tge_request_recalc_owner = yes

	# A factory keeps its finished good if something else changes the trade good
	# Fail-safe: change_trade_goods fires this on_action again right away; the flag stops a second round
	if = {
		limit = { NOT = { has_province_flag = tge_enforcing_good } }
		set_province_flag = tge_enforcing_good
		tge_enforce_factory_good = { BUILDING = cannon_foundry FLAG = has_cannon GOOD = cannon }
		tge_enforce_factory_good = { BUILDING = metalworking_workshop FLAG = has_metalworking GOOD = metalworking }
		tge_enforce_factory_good = { BUILDING = steam_engine_manu FLAG = has_steam_engine GOOD = steam_engine }
		tge_enforce_factory_good = { BUILDING = spectacle_factory FLAG = has_optics GOOD = optics }
		tge_enforce_factory_good = { BUILDING = munition_factory FLAG = has_munitions GOOD = munitions }
		clr_province_flag = tge_enforcing_good
	}
}

# ROOT = province, FROM = old owner
on_province_owner_change = {
	# Strategic resources: both sides recalculate (factories and output follow the province)
	if = {
		limit = { tge_is_resource_relevant_province = yes }
		owner = { tge_request_recalc = yes }
		FROM = { tge_request_recalc = yes }
	}
	# Racial building bonuses
	tge_racial_bonus_refresh_province = yes
}
